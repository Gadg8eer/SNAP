namespace = SNAP_CheckOntology

# Game start
event = {
	id = SNAP_CheckOntology.0
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		# Unless otherwise specified by the creator, personas/characters who are minors will not die.
		if = {
			limit = {
				OR = {
					AND = {
						leader_age < 18
						NOT = {
							has_trait = leader_trait_snap_protected_youth
						}
					}
					AND = {
						leader_age < 18
						NOT = {
							has_trait = leader_trait_snap_protected_youth
						}
						OR = {
							has_trait = leader_trait_SNAP_CodyGibling_Gadg8eer
							has_trait = leader_trait_SNAP_CodyGibling_quick_Gadg8eer
							has_trait = leader_trait_SNAP_CodyGibling_social_Gadg8eer
					}
					AND = {
						leader_age < 18
						NOT = {
							has_trait = leader_trait_snap_protected_youth
						}
						OR = {
							has_trait = leader_trait_SNAP_Steven_Wintle_Jenny_Everywhere
							has_trait = leader_trait_SNAP_Steven_Wintle_Jenny_Everywhere_2
						}
					}
					AND = {
						NOT = {
							has_trait = leader_trait_snap_protected_youth
						}
						leader_age < 18
						OR = {
							has_trait = leader_trait_SNAP_MisterEightySix_ArcoIris
						}
					}
					AND = {
						NOT = {
							has_trait = leader_trait_snap_protected_youth
						}
						has_trait = leader_trait_SNAP_CodyGibling_playing_Gadg8eer
					}
				}
			}
		}
		
		fire_event = {
			id = SNAP_CheckOntology.1
		}
	}
}

# January 2nd, start year
event = {
	id = SNAP_CheckOntology.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = no

	immediate = {
		every_country = {
			limit = {
				# Only human players
				is_ai = no
			}
			
			# Check for ArcoIris
			if = {
				limit = {
					any_owned_leader = {
						AND = {
							leader_age < 18
							OR = {
								has_trait = leader_trait_SNAP_MisterEightySix_ArcoIris
							}
						}
					}
				}
				add_trait = leader_trait_snap_protected_youth
				fire_event = {
					id = SNAP_events_MisterEightySix_ArcoIris.0
				}
			}

			# Check for Jenny Everywhere
			else_if = {
				limit = {
					any_owned_leader = {
						AND = {
							leader_age < 18
							OR = {
								has_trait = leader_trait_SNAP_Steven_Wintle_Jenny_Everywhere
								has_trait = leader_trait_SNAP_Steven_Wintle_Jenny_Everywhere_2
							}
						}
					}
				}
				add_trait = leader_trait_snap_protected_youth
				fire_event = {
					id = SNAP_events_Steven_Wintle_Jenny_Everywhere.0
				}
			}

			# Check for Gadg8eer
			else_if = {
				limit = {
					any_owned_leader = {
						AND = {
							leader_age < 18
							OR = {
								has_trait = leader_trait_SNAP_CodyGibling_Gadg8eer
								has_trait = leader_trait_SNAP_CodyGibling_quick_Gadg8eer
								has_trait = leader_trait_SNAP_CodyGibling_social_Gadg8eer
							}
						}
					}
				}
				fire_event = {
					id = SNAP_events_CodyGibling_Gadg8eer.0
				}
			} else_if = {
				limit = {
					any_owned_leader = {
						AND = {
							has_trait = leader_trait_snap_protected_youth
							OR = {
								has_trait = leader_trait_SNAP_CodyGibling_playing_Gadg8eer
							}
						}
					}
				}
				fire_event = {
					id = SNAP_events_CodyGibling_Gadg8eer.10
				}
			}
		}
		# Unless otherwise specified by the creator, personas/characters who are minors will not die.
		if = {
			limit = {
				AND = {
					# Only human players
					is_ai = no
				}
				NOT = {
					is_immortal = yes
					leader_age >= 18
				}
				OR = {	
					has_trait = leader_trait_SNAP_CodyGibling_playing_Gadg8eer
				}
			}
			remove_trait = leader_trait_snap_protected_youth
		}
	}
}
