### Start with Gadg8eer
SNAP_scripted_effect_CodyGibling_Gadg8eer_start = {
	has_trait = trait_ruler_SNAP_CodyGibling_Gadg8eer # Check if the leader actually *has* the trait
	immediate = {
		create_leader = {
			class = scientist
			species = root.species
			name = leader_CodyGibling_Gadg8eer
			gender = male
			skill = 3
			traits = {
				trait = leader_trait_SNAP_CodyGibling_Gadg8eer
			}
			owner = root
			save_event_target_as = SNAP_CodyGibling_Gadg8eer
		}

		event_target:SNAP_CodyGibling_Gadg8eer = {
			set_age = 8
			set_ruler = yes
		}
	
	# Set flag because THERE CAN BE ONLY ONE!
		set_global_flag = global_SNAP_rulerclaim_CodyGibling_Gadg8eer
	}
}


### Start with (personalized, kinda cheat-y) Gadg8eer
SNAP_scripted_effect_CodyGibling_playing_Gadg8eer_start = {
	has_trait = trait_ruler_SNAP_CodyGibling_playing_Gadg8eer # Check if the leader actually *has* the trait
	immediate = {
		create_leader = {
			class = scientist
			species = root.species
			name = leader_CodyGibling_Gadg8eer
			gender = male
			skill = 3
			traits = {
				trait = leader_trait_SNAP_CodyGibling_playing_Gadg8eer
				trait = trait_mechanical
			}
			owner = root
			save_event_target_as = SNAP_CodyGibling_Gadg8eer
		}

		event_target:SNAP_CodyGibling_Gadg8eer = {
			set_portrait = "SNAP_portraits_CodyGibling_Gadg8eer"
			set_age = 10
			set_immortal = yes
			set_ruler = yes
		}
		
	# Set flag because THERE CAN BE ONLY ONE!
		set_global_flag = global_SNAP_rulerclaim_CodyGibling_Gadg8eer
	}
}

### If Gadg8eer is manually removed from the position of ruler (by the player), he is returned to the position and a message is displayed.
SNAP_scripted_effect_CodyGibling_Gadg8eer_oops_noremove = {
	#Consider changing this to an on_action trigger if possible for more reliability
	pre_triggers = {
		has_owner = yes		# whether the leader exists and is alive
		is_idle = yes		# whether the leader is idle
	}
	trigger = {
		OR = {
			has_trait = trait_ruler_SNAP_CodyGibling_Gadg8eer # Check if the leader actually *has* the trait
			is_same_value = {
				first = root
				second = event_target:SNAP_CodyGibling_Gadg8eer
			}
		}
	}
	immediate = {
		SNAP_CodyGibling_Gadg8eer = { #Check if ruler before attempting assignement
				limit = {
					is_valid_leader_for_position = ruler
				}
				assign_leader = ruler
				set_leader_flag = prevent_remove_SNAP_CodyGibling_Gadg8eer
				leader_event = {
					id = SNAP_events_CodyGibling_Gadg8eer.3
				}
			}
		}
	}
}

### If Gadg8eer becomes 86 years old or is forcibly removed from the position of ruler, he dies.
### Also includes compatibility with Constructorvette Mod.
SNAP_scripted_effect_codygibling_death_of_an_inner_child = {
	event = {
	#Consider changing this to an on_action trigger if possible for more reliability
		pre_triggers = {
			has_owner = yes		# whether the leader exists and is alive
			is_idle = yes		# whether the leader is idle
		}
	trigger = {
		AND = {
			OR = {
				has_trait = trait_ruler_SNAP_CodyGibling_Gadg8eer # Check if the leader actually *has* the trait
				is_same_value = {
					first = root
					second = event_target:SNAP_CodyGibling_Gadg8eer
				}
			}
			age = 86
		}
	}
}
